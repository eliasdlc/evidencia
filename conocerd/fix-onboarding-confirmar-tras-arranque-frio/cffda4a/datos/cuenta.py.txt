#!/usr/bin/env python3
"""cuenta.py EMAIL: prints the state of users/{uid} of a throwaway account
(profilePreferences.status, welcomeStamp present, createdAt present) and no
credential. The password comes from ~/.config/secretos/conocerd-hilo.env, the
web API key from android/app/google-services.json, the read token from gcloud."""
import json, pathlib, subprocess, sys, urllib.request
email = sys.argv[1]
home = pathlib.Path.home()
secretos = (home / ".config/secretos/conocerd-hilo.env").read_text().splitlines()
clave = next(l.split("=", 1)[1] for l in secretos if l.startswith(email + "="))
gs = json.loads((home / "dev/wt/conocerd/fix/onboarding-confirmar-tras-arranque-frio/android/app/google-services.json").read_text())
api = gs["client"][0]["api_key"][0]["current_key"]
req = urllib.request.Request(
    f"https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key={api}",
    data=json.dumps({"email": email, "password": clave, "returnSecureToken": True}).encode(),
    headers={"Content-Type": "application/json", "X-Android-Package": "com.conocerd.app"},
)
try:
    uid = json.load(urllib.request.urlopen(req, timeout=20))["localId"]
except Exception as e:
    print("sign-in failed:", type(e).__name__, getattr(e, "code", "")); sys.exit(1)
tok = subprocess.run(["gcloud", "auth", "print-access-token"], capture_output=True, text=True).stdout.strip()
url = f"https://firestore.googleapis.com/v1/projects/conocerd-dev/databases/(default)/documents/users/{uid}"
r = urllib.request.Request(url, headers={"Authorization": f"Bearer {tok}", "x-goog-user-project": "conocerd-dev"})
try:
    doc = json.load(urllib.request.urlopen(r, timeout=20))
except Exception as e:
    print("read failed:", type(e).__name__, getattr(e, "code", "")); sys.exit(1)
f = doc.get("fields", {})
pp = f.get("profilePreferences", {}).get("mapValue", {}).get("fields", {})
status = pp.get("status", {}).get("stringValue")
ws = f.get("welcomeStamp", {}).get("mapValue", {}).get("fields", {})
print(f"users/<uid {uid[:6]}...> updateTime {doc.get('updateTime')} profilePreferences.status={status} "
      f"welcomeStamp={'granted ' + ws.get('grantedAt', {}).get('timestampValue', '?') if ws else 'absent'} "
      f"createdAt={'present' if 'createdAt' in f else 'absent'}")
